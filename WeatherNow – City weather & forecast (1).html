Show real OpenWeather data

A published page can’t call the API itself. Call OpenWeather (the Python code does this), then paste the JSON here. The page reads it exactly like Python’s response.json(). It accepts the Current weather response (/weather) or the 5-day forecast response (/forecast), fetched with units=metric.

The example uses made-up numbers to show the format. It is not live weather.

Python app (requests + JSON, runs in the terminal)

Get a free key at openweathermap.org, run pip install requests, paste your key, and run the file.

import requests
from collections import defaultdict

API_KEY = "YOUR_API_KEY"
BASE = "https://api.openweathermap.org/data/2.5"

def get(path, city):
    r = requests.get(f"{BASE}/{path}", timeout=10,
                     params={"q": city, "appid": API_KEY, "units": "metric"})
    if r.status_code == 404:
        raise SystemExit("City not found. Check the spelling.")
    r.raise_for_status()
    return r.json()

city = input("City name: ").strip()
now = get("weather", city)

print(f"\nWeather in {now['name']}")
print(f"Temperature : {now['main']['temp']} C")
print(f"Humidity    : {now['main']['humidity']} %")
print(f"Condition   : {now['weather'][0]['description'].title()}")
print(f"Wind speed  : {round(now['wind']['speed'] * 3.6, 1)} km/h")

days = defaultdict(list)
for item in get("forecast", city)["list"]:
    days[item["dt_txt"][:10]].append(item)

print("\nNext days")
for date, items in list(days.items())[:5]:
    temps = [i["main"]["temp"] for i in items]
    rain = max(i.get("pop", 0) for i in items) * 100
    cond = items[len(items) // 2]["weather"][0]["main"]
    print(f"{date}: {min(temps):.0f}-{max(temps):.0f} C, {cond}, rain {rain:.0f}%")
